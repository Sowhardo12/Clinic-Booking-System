

import { Injectable } from '@nestjs/common';
import { DatabaseService } from '../database/database.service';
import { PoolClient } from 'pg';

@Injectable()
export class AppointmentsRepository {
  constructor(private db: DatabaseService) {}

  async getClient(): Promise<PoolClient> {
    return this.db.getClient();
  }

  async findSlotByIdForUpdate(client: PoolClient, slotId: number) {
    const query = `SELECT * FROM appointment_slots WHERE id = $1 FOR UPDATE`;
    const result = await client.query(query, [slotId]);
    return result.rows[0];
  }

  async issueTicket(client: PoolClient, slotId: number, patientId: number, testRunId?: string) {
    const query = `
      INSERT INTO patient_tickets (slot_id, patient_id, test_run_id)
      VALUES ($1, $2, $3)
      RETURNING *;
    `;
    const result = await client.query(query, [slotId, patientId, testRunId || null]);
    return result.rows[0];
  }

  async bookSlot(client: PoolClient, slotId: number, patientId: number) {
    const query = `
      UPDATE appointment_slots 
      SET status = 'booked', patient_id = $1 
      WHERE id = $2 
      RETURNING *;
    `;
    const result = await client.query(query, [patientId, slotId]);
    return result.rows[0];
  }

  async logBookingAttempt(client: PoolClient, slotId: number, patientId: number, testRunId: string, status: string) {
    await client.query(
      `INSERT INTO booking_attempts (slot_id, patient_id, test_run_id, attempt_status) VALUES ($1, $2, $3, $4)`,
      [slotId, patientId, testRunId || null, status]
    );
  }
}
